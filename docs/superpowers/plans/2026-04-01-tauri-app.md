```rust
fn toggle_panel(app: &tauri::AppHandle) {
    if let Some(window) = app.get_webview_window(PANEL_LABEL) {
        // Panel exists — toggle visibility
        if window.is_visible().unwrap_or(false) {
            let _ = window.hide();
        } else {
            // Reposition at tray before showing
            let _ = window.as_ref().window().move_window(Position::TrayCenter);
            let _ = window.show();
            let _ = window.set_focus();
        }
    } else {

        let url = WebviewUrl::App("index.html?layout=menubar".into());

            .transparent(true)
            .always_on_top(true)
            .visible(false)
            .skip_taskbar(true)

                        if let Some(w) = app_handle.get_webview_window(PANEL_LABEL) {
                            let _ = w.hide();
                        }
                    }
                });
            }
            Err(e) => {
                eprintln!("Failed to create panel window: {}", e);
            }
        }
    }
```

## Task 5: API Sidecar Management

The Tauri app needs to ensure the kennel API server is running on `localhost:5544`. Strategy:

1. On launch, check if port 5544 is already responding (user ran `kennel ui` manually)
2. If not, spawn the kennel API server as a child process
3. Wait for it to be healthy before loading the frontend
4. Kill the child process on app exit

We use `std::process::Command` from Rust directly (not the shell plugin) because this is internal app lifecycle management, not user-facing shell access. The API server is `node dist/cli.js ui --port 5544`.

- [ ] Add `api.rs` module for API server lifecycle management
- [ ] Implement health check (`GET /api/services` returns 200)
- [ ] Implement spawn logic with `std::process::Command`
- [ ] Implement detection of existing instance
- [ ] Wire into app setup — start API before building tray
- [ ] Kill child process on app exit
- [ ] Add Tauri command so frontend can check API status

### src-tauri/src/api.rs

Create this new file:

```rust
use std::process::{Child, Command};
use std::sync::Mutex;
use std::time::Duration;

const API_HOST: &str = "http://localhost";
const API_PORT: u16 = 5544;
const HEALTH_ENDPOINT: &str = "/api/services";
const MAX_STARTUP_WAIT: Duration = Duration::from_secs(10);
const POLL_INTERVAL: Duration = Duration::from_millis(200);

pub struct ApiServer {
    child: Mutex<Option<Child>>,
    port: u16,
}

impl ApiServer {
    pub fn new(port: u16) -> Self {
        Self {
            child: Mutex::new(None),
            port,
        }
    }

    /// Check if the API server is already running on the configured port
    pub async fn is_running(&self) -> bool {
        let url = format!("{}:{}{}", API_HOST, self.port, HEALTH_ENDPOINT);
        match reqwest::Client::new()
            .get(&url)
            .timeout(Duration::from_secs(2))
            .send()
            .await
        {
            Ok(resp) => resp.status().is_success(),
            Err(_) => false,
        }
    }

    /// Start the kennel API server as a child process.
    /// Looks for `node` and the kennel CLI in standard locations.
    pub fn spawn(&self) -> Result<(), String> {
        // Find the kennel CLI entry point
        // In dev: use tsx to run from source
        // In production: use node to run compiled JS
        let (program, args) = Self::resolve_command(self.port)?;

        let child = Command::new(&program)
            .args(&args)
            .stdout(std::process::Stdio::piped())
            .stderr(std::process::Stdio::piped())
            .spawn()
            .map_err(|e| format!("Failed to spawn API server: {}", e))?;

        let mut guard = self.child.lock().unwrap();
        *guard = Some(child);
        Ok(())
    }

    /// Wait for the API server to become healthy
    pub async fn wait_for_healthy(&self) -> Result<(), String> {
        let start = std::time::Instant::now();
        while start.elapsed() < MAX_STARTUP_WAIT {
            if self.is_running().await {
                return Ok(());
            }
            tokio::time::sleep(POLL_INTERVAL).await;
        }
        Err("API server did not become healthy within timeout".to_string())
    }

    /// Ensure the API server is running — detect existing or spawn new
    pub async fn ensure_running(&self) -> Result<bool, String> {
        // Check if already running (user started `kennel ui` manually)
        if self.is_running().await {
            return Ok(false); // false = we didn't spawn it, it was already running
        }

        // Spawn a new instance
        self.spawn()?;
        self.wait_for_healthy().await?;
        Ok(true) // true = we spawned it
    }

    /// Kill the child process if we spawned it
    pub fn shutdown(&self) {
        let mut guard = self.child.lock().unwrap();
        if let Some(ref mut child) = *guard {
            let _ = child.kill();
            let _ = child.wait();
        }
        *guard = None;
    }

    /// Resolve the command to start the API server
    fn resolve_command(port: u16) -> Result<(String, Vec<String>), String> {
        let port_str = port.to_string();

        // Strategy 1: Check if `kennel` is in PATH (installed globally via npm)
        if let Ok(output) = Command::new("which").arg("kennel").output() {
            if output.status.success() {
                let kennel_path = String::from_utf8_lossy(&output.stdout).trim().to_string();
                return Ok((kennel_path, vec!["ui".into(), "--port".into(), port_str]));
            }
        }

        // Strategy 2: Look for dist/cli.js relative to the app bundle
        // In a bundled .app, resources are at Contents/Resources/
        // We also check relative to the executable for dev builds
        let exe_dir = std::env::current_exe()
            .ok()
            .and_then(|p| p.parent().map(|p| p.to_path_buf()));

        if let Some(dir) = &exe_dir {
            // Check for bundled CLI
            let cli_path = dir.join("../Resources/dist/cli.js");
            if cli_path.exists() {
                return Ok((
                    "node".into(),
                    vec![
                        cli_path.to_string_lossy().to_string(),
                        "ui".into(),
                        "--port".into(),
                        port_str,
                    ],
                ));
            }
        }

        // Strategy 3: Use npx kennel (development fallback)
        Ok(("npx".into(), vec!["kennel".into(), "ui".into(), "--port".into(), port_str]))
    }

    pub fn port(&self) -> u16 {
        self.port
    }

    pub fn base_url(&self) -> String {
        format!("{}:{}", API_HOST, self.port)
    }
}

impl Drop for ApiServer {
    fn drop(&mut self) {
        self.shutdown();
    }
}
```

```rust
            .title("Kennel")
            .inner_size(PANEL_WIDTH, PANEL_HEIGHT)
            .resizable(false)
            .decorations(false)
            .transparent(true)
            .always_on_top(true)
            .visible(false)
            .skip_taskbar(true)
            .build();

        match panel {
            Ok(window) => {
                let _ = window.as_ref().window().move_window(Position::TrayCenter);
                let _ = window.show();
                let _ = window.set_focus();

                let app_handle = app.clone();
                window.on_window_event(move |event| {
                    if let tauri::WindowEvent::Focused(false) = event {
                        if let Some(w) = app_handle.get_webview_window(PANEL_LABEL) {
                            let _ = w.hide();
                        }
                    }
                });
            }
            Err(e) => {
                eprintln!("Failed to create panel window: {}", e);
            }
        }
    }
}
```

### Updated src-tauri/src/api.rs (with Clone support)

Change the struct to use `Arc<Mutex<>>` internally and derive `Clone`:

```rust
use std::process::{Child, Command};
use std::sync::{Arc, Mutex};
use std::time::Duration;

const API_HOST: &str = "http://localhost";
const HEALTH_ENDPOINT: &str = "/api/services";
const MAX_STARTUP_WAIT: Duration = Duration::from_secs(10);
const POLL_INTERVAL: Duration = Duration::from_millis(200);

#[derive(Clone)]
pub struct ApiServer {
    child: Arc<Mutex<Option<Child>>>,
    port: u16,
}

impl ApiServer {
    pub fn new(port: u16) -> Self {
        Self {
            child: Arc::new(Mutex::new(None)),
            port,
        }
    }

    pub async fn is_running(&self) -> bool {
        let url = format!("{}:{}{}", API_HOST, self.port, HEALTH_ENDPOINT);
        match reqwest::Client::new()
            .get(&url)
            .timeout(Duration::from_secs(2))
            .send()
            .await
        {
            Ok(resp) => resp.status().is_success(),
            Err(_) => false,
        }
    }

    pub fn spawn(&self) -> Result<(), String> {
        let (program, args) = Self::resolve_command(self.port)?;

        let child = Command::new(&program)
            .args(&args)
            .stdout(std::process::Stdio::piped())
            .stderr(std::process::Stdio::piped())
            .spawn()
            .map_err(|e| format!("Failed to spawn API server: {}", e))?;

        let mut guard = self.child.lock().unwrap();
        *guard = Some(child);
        Ok(())
    }

    pub async fn wait_for_healthy(&self) -> Result<(), String> {
        let start = std::time::Instant::now();
        while start.elapsed() < MAX_STARTUP_WAIT {
            if self.is_running().await {
                return Ok(());
            }
            tokio::time::sleep(POLL_INTERVAL).await;
        }
        Err("API server did not become healthy within timeout".to_string())
    }

    pub async fn ensure_running(&self) -> Result<bool, String> {
        if self.is_running().await {
            return Ok(false);
        }
        self.spawn()?;
        self.wait_for_healthy().await?;
        Ok(true)
    }

    pub fn shutdown(&self) {
        let mut guard = self.child.lock().unwrap();
        if let Some(ref mut child) = *guard {
            let _ = child.kill();
            let _ = child.wait();
        }
        *guard = None;
    }

    fn resolve_command(port: u16) -> Result<(String, Vec<String>), String> {
        let port_str = port.to_string();

        // Strategy 1: `kennel` in PATH
        if let Ok(output) = Command::new("which").arg("kennel").output() {
            if output.status.success() {
                let kennel_path = String::from_utf8_lossy(&output.stdout).trim().to_string();
                return Ok((kennel_path, vec!["ui".into(), "--port".into(), port_str]));
            }
        }

        // Strategy 2: Bundled dist/cli.js in .app bundle
        let exe_dir = std::env::current_exe()
            .ok()
            .and_then(|p| p.parent().map(|p| p.to_path_buf()));

        if let Some(dir) = &exe_dir {
            let cli_path = dir.join("../Resources/dist/cli.js");
            if cli_path.exists() {
                return Ok((
                    "node".into(),
                    vec![
                        cli_path.to_string_lossy().to_string(),
                        "ui".into(),
                        "--port".into(),
                        port_str,
                    ],
                ));
            }
        }

        // Strategy 3: npx fallback (dev)
        Ok(("npx".into(), vec!["kennel".into(), "ui".into(), "--port".into(), port_str]))
    }

    pub fn port(&self) -> u16 {
        self.port
    }

    pub fn base_url(&self) -> String {
        format!("{}:{}", API_HOST, self.port)
    }
}

impl Drop for ApiServer {
    fn drop(&mut self) {
        self.shutdown();
    }
}
```

If no source icon exists yet, create a placeholder programmatically:

```bash
# Create a simple placeholder 1024x1024 icon
python3 -c "
import struct, zlib

def create_png(width, height, color=(41, 98, 255)):
    def chunk(ty, data):
        c = ty + data
        return struct.pack('>I', len(data)) + c + struct.pack('>I', zlib.crc32(c) & 0xffffffff)

    raw = b''
    r, g, b = color
    for y in range(height):
        raw += b'\x00'  # filter byte
        for x in range(width):
            # Simple rounded square with K letter
            margin = width // 8
            corner_r = width // 6
            in_square = margin <= x < width - margin and margin <= y < height - margin

            if in_square:
                # Check corners for rounding
                corners = [
                    (margin + corner_r, margin + corner_r),
                    (width - margin - corner_r, margin + corner_r),
                    (margin + corner_r, height - margin - corner_r),
                    (width - margin - corner_r, height - margin - corner_r),
                ]
                in_corner = False
                for cx, cy in corners:
                    dx = abs(x - cx)
                    dy = abs(y - cy)
                    if (x < margin + corner_r or x >= width - margin - corner_r) and \
                       (y < margin + corner_r or y >= height - margin - corner_r):
                        if dx * dx + dy * dy > corner_r * corner_r:
                            in_corner = True

                if not in_corner:
                    raw += bytes([r, g, b, 255])
                else:
                    raw += b'\x00\x00\x00\x00'
            else:
                raw += b'\x00\x00\x00\x00'

    sig = b'\x89PNG\r\n\x1a\n'
    ihdr = struct.pack('>IIBBBBB', width, height, 8, 6, 0, 0, 0)
    return sig + chunk(b'IHDR', ihdr) + chunk(b'IDAT', zlib.compress(raw, 9)) + chunk(b'IEND', b'')

# Write the base icon
with open('assets/app-icon.png', 'wb') as f:
    f.write(create_png(1024, 1024))
print('Created placeholder app-icon.png')
"

# Then generate all sizes
cd src-tauri && cargo tauri icon ../assets/app-icon.png
```
